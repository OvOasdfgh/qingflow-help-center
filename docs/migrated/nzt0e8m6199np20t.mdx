## 场景描述

使用轻流系统实现会议室预约场景，满足会议室被占用期间不可被预约，以及取消后需要将该会议室该时段需要被释放出来。

**预约会议室实现要求**

1、会议室被占用期间不可被预约

2、取消后需要将该会议室该时段需要被释放出来

3、会议室预约时间至少30分钟起步（可修改）

4、会议室预约时间不可预约之前时间

5、会议室预约时间需要以5分钟为精度（可修改）

## 思路逻辑

具体使用代码块来完成遇上预约会议室要求。其中需要调用OpenApi获取自身应用数据的会议室以及预约会议室起止时间，然后拿发起新数据的会议室以及预约会议室起止时间来跟自身应用的数据进行对比。

如果当前会议室的预约时间跟自身应用数据的历史会议时间有冲突就会无法提交预约

​

注意：因为是跟自身应用做比较的，所以可能会存在并发的情况。所以请避免同时发起预约同一时段。

后续代码块节点功能上线可解决。

const axios = require('axios');

function isOverlap(date, startTime, endTime) {
    const [start, end] = date.split('~');
  
    if (startTime <= start && start <= endTime) {
      return true;
    }
  
    if (startTime <= end && end <= endTime) {
      return true;
    }
  
    if (start <= startTime && endTime <= end) {
      return true;
    }
  
    return false;
  }
  
  function checkAvailability(date, room, reslist) {
    const filteredResList = reslist.filter(item => item['会议室'] === room);
  
    for (const item of filteredResList) {
      const [reservationStart, reservationEnd] = item['预约会议起止时间'].split('~');
      const overlap = isOverlap(date, reservationStart, reservationEnd);
  
      if (overlap) {
        return false; // 时间区间已被覆盖
      }
    }
  
    return true; // 时间区间可用
  }
  
  

// 调用接口获取reslist
async function fetchData() {
  
    const data = JSON.stringify({
      "pageSize": "50",
      "pageNum": "1"
    });

    const config = {
      method: 'post',
      url: 'https://api.qingflow.com/app/16365e01/apply/filter',
      headers: {
        'accessToken': 'd866db1d-79ca-42c5-9e9c-0b1ce21135d2',
        'Content-Type': 'application/json'
      },
      data: data
    };

    const res = await axios(config);
    const msg = res.data;
    const result = msg.result.result;

    const reslist = result.map(value => {
      const res_msg = {};

      value.answers.forEach(v => {
        const queId = v.queId;
        const queTitle = v.queTitle;

        if (queId === 166744302 || queId === 166753788) {
          const values = v.values;
          const dataValue = values[0].dataValue;
          res_msg[queTitle] = dataValue;
        }
      });

      return res_msg;
    });

    const date = qf_field.{预约会议起止时间$$178A98D01$$}; // 添加一个判断条件，若为null，则isAvailable为true
    const room = qf_field.{会议室$$178A8997A$$};
    const isAvailable = (date === null) ? true : await checkAvailability(date, room, reslist);
    qf_output={isAvailable}
    //console.log(isAvailable);
  
}

function validateTime(time) {
    // 分割起止时间
    const [startTime, endTime] = time.split("~");
    
    // 提取起始时间的时分部分
    const [startHour, startMinute] = startTime.split(" ")[1].split(":");
    
    // 提取结束时间的时分部分
    const [endHour, endMinute] = endTime.split(" ")[1].split(":");
    
    // 将时分转换为分钟
    const startTotalMinutes = parseInt(startHour) * 60 + parseInt(startMinute);
    const endTotalMinutes = parseInt(endHour) * 60 + parseInt(endMinute);
    
    // 校验条件：时分换算成分之后，除以5能被整除，时分为00:00的话除外
    if (startTotalMinutes === 0 && endTotalMinutes === 0) {
      return true; // 时分为00:00，通过校验
    } else {
      return (startTotalMinutes % 5 === 0) && (endTotalMinutes % 5 === 0);
    }
  }
  
  // 测试校验函数
  const date = qf_field.{预约会议起止时间$$178A98D01$$};
  const isValid = (date === null) ? true: validateTime(date);
  qf_output={isValid}
  //console.log(isValid); // 输出 true 或者 false

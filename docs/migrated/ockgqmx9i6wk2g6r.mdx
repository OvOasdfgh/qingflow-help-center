## 场景/问题描述

方案：[表格序号如何自动生成？](https://exiao.yuque.com/enzu8l/oy1eh8/hkl3rq)

在 表格序号自动生成 方案中，有个方案用到了多行文本拼接随机码以实现查询并生成序号，但客户实际业务遇到行数特别多的情形，5000个字符的限制使得客户在这类场景下没办法使用这个方案。

## 思路逻辑

通过代码块缩短随机码，实现一一对应的唯一码的方式来增加客户可用行数

### 代码块配置

```javascript
var table = qf_field.{表格$$18B5AA9BD$$};

// 生成短网址，随机码唯一，生成的短随机码也唯一
function generateShortURL() {
  var chars = 'abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789';
  var shortURL = '';
  var length = 6; // 要生成的短码长度

  for (var i = 0; i < length; i++) {
    var randomIndex = Math.floor(Math.random() * chars.length);
    shortURL += chars.charAt(randomIndex);
  }

  return shortURL;
}

// 更新数组中的shortURL属性
function updateShortURL(table) {
  for (var i = 0; i < table.length; i++) {
    var randomCode = table[i].randomCode;
    var shortURL = generateShortURL();
    table[i].field_179265346 = shortURL;
  }
}

// 调用函数更新shortURL属性
updateShortURL(table);

qf_output = {r : table}
```

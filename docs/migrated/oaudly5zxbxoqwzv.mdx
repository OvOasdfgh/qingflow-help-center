### 扫码枪扫码自动录入到表格中

- 扫码的内容需要连续录入到单行文字中，并且中间用逗号分隔。
- 目的是为了连续扫码并且代码块自动运行
- inputString为插入表单的单行文字字段

```javascript
function processString(inputString) {
    // 去除字符串首尾的空格和逗号
    const trimmedString = inputString.trim().replace(/(^,)|(,$)/g, '');

    // 使用逗号分割字符串来形成一个数组
    const items = trimmedString.split(',');

    // 过滤掉空字符串，然后使用Set对象去除数组中的重复项
    const uniqueItems = Array.from(new Set(items.filter(item => item.trim() !== '')));

    // 将结果数组中的每个值封装成一个对象，键为'value'
    const result = uniqueItems.map(item => ({ value: item.trim() }));

    // 返回处理后的数组对象
    return result;
}

}else{
  // 处理并输出结果
const processedData = processString(inputString);
//console.log(processedData);

}
```

```javascript
const axios = require('axios');
const xlsx = require('node-xlsx');
const { format, addDays } = require('date-fns');

/**
* 读取轻流系统Excel附件中指定区域的数据
* @param {string} qflowUrl 轻流附件URL
* @param {number} startRow 起始行索引
* @param {number} endRow 结束行索引（不包含）
* @param {number} startCol 起始列索引
* @param {number} endCol 结束列索引（不包含）
* @returns {Promise<object[]>} Excel指定区域数据的JSON对象数组
*/
async function getExcelData(qflowUrl, startRow, endRow, startCol, endCol) {
  const response = await axios.get(qflowUrl, { responseType: 'arraybuffer' });
  const excel = xlsx.parse(response.data);
  const data = excel[0].data;

  const headers = data[startRow].slice(startCol, endCol); // 提取表头数据
  const rows = data.slice(startRow + 1, endRow); // 提取行数据

  const jsonData = rows.map(row => {
    const obj = {};
    headers.forEach((header, index) => {
      const value = row[index + startCol];
      if (header === "birth") {
        const birthDate = addDays(new Date(1900, 0, 1), value - 1); // 使用date-fns将序列数转换为日期
        obj[header] = format(birthDate, 'yyyy-MM-dd'); // 使用date-fns格式化日期为字符串
      } else {
        obj[header] = value;
      }
    });
    return obj;
  });

  return jsonData;
}
```

```javascript
const axios = require('axios');
const csv = require('csv-parser');
const { Transform } = require('stream');

async function fetchCSV(url) {
  try {
    const response = await axios({
      method: 'get',
      url: url,
      responseType: 'stream'
    });

    const data = [];
    const parser = response.data.pipe(csv());

    parser.on('data', (row) => {
      data.push(row);
```

```javascript
function encode(text, secretKey) {

  const cipher = crypto.createCipheriv('des-ede3-cbc', secretKey, Buffer.alloc(8));

  let decrypted = decipher.update(encryptedBase64, 'base64', 'utf8');
  decrypted += decipher.final('utf8');
  return decrypted;
};
```

```javascript
        res_msg[queIdTitleMap[queId]] = dataValue; // 使用映射中的queTitle

  });

    });
```
```javascript
  });
```

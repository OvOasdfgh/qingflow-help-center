## 场景/问题描述

需要按照分类为中干以及改任来进行重组数组，并且按照4行为一组进行重组为新的一行。以便于打印数据时，减少打印纸张

## 解决思路

通过代码块先根据类别提取新的数组，然后再根据数组长度进行4行为一组进行重组，最后进行输出

    function groupItems(items) {
        let grouped = [];
        for (let i = 0; i < items.length; i += 4) {
            let group = {};
            for (let j = 0; j < 4; j++) {

                group["评分" + j] = items[i + j] ? items[i + j].评分 : null;
            }
            grouped.push(group);
        }
        return grouped;
    }

    // 处理后的数组
    let list1 = groupItems(zhongan);
    let list2 = groupItems(gairen);

}
